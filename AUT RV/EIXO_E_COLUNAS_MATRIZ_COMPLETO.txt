================================================================================
EIXO_COLABORADORES + COLUNAS CALCULADAS - ESTRUTURA MATRIZ
================================================================================
Objetivo: Matriz → Cargo (Coordenador/Líder/Assistente) → Nome
          Fora Matriz: Coordenador → Líder → Polo → Cargo → Nome

Dependências: DimColaboradores, FactOS, FactCSAT
================================================================================

--------------------------------------------------------------------------------
1. EIXO_COLABORADORES (Tabela calculada)
--------------------------------------------------------------------------------

Eixo_Colaboradores = 
VAR _ops_dim =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(
                DimColaboradores,
                "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" ),
                "__EhMatrizPolo",
                    VAR x = UPPER(TRIM(COALESCE(DimColaboradores[Polo_Base], "")))
                    RETURN x = "MATRIZ" || x = "PAGRESOLVE MATRIZ SP"
            ),
            [__CargoNorm] IN {"TECNICO","ASSISTENTE"}
        ),
        "Nome",                DimColaboradores[Nome],
        "Nome_Normalizado",
            VAR n = UPPER(TRIM(DimColaboradores[Nome]))
            VAR n1 = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n,
                    "Á","A"),"À","A"),"Â","A"),"Ã","A"),"É","E"),"Ê","E"),"Í","I"),"Ó","O"),"Ô","O"),"Õ","O")
            RETURN SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n1,"Ú","U"),"Ü","U"),"Ç","C"),
        "Cargo",               DimColaboradores[Cargo],
        "Matricula",           DimColaboradores[Matricula],
        "Polo_Base",           DimColaboradores[Polo_Base],
        "Polo Base (Regra)",   IF([__EhMatrizPolo], "Matriz", DimColaboradores[Polo_Base]),
        "IsMatriz",            [__EhMatrizPolo],
        "Grupo",               "Operação",
        "Fonte",               "DIM"
    )

VAR _gest_matriz_dim =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(
                DimColaboradores,
                "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" ),
                "__EhMatriz",
                    VAR _raw   = FORMAT(DimColaboradores[IsMatriz], "")
                    VAR _norm  = UPPER(TRIM(COALESCE(_raw, "")))
                    VAR _txt   = _norm IN {"TRUE","VERDADEIRO","SIM","YES","1"}
                    VAR _polo  = UPPER(TRIM(COALESCE(DimColaboradores[Polo_Base], "")))
                    RETURN _txt || _polo = "MATRIZ" || _polo = "PAGRESOLVE MATRIZ SP"
            ),
            [__CargoNorm] IN {"LIDER","COORDENADOR"} && [__EhMatriz]
        ),
        "Nome",                DimColaboradores[Nome],
        "Nome_Normalizado",
            VAR n2 = UPPER(TRIM(DimColaboradores[Nome]))
            VAR n2a = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n2,
                    "Á","A"),"À","A"),"Â","A"),"Ã","A"),"É","E"),"Ê","E"),"Í","I"),"Ó","O"),"Ô","O"),"Õ","O")
            RETURN SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n2a,"Ú","U"),"Ü","U"),"Ç","C"),
        "Cargo",               DimColaboradores[Cargo],
        "Matricula",           DimColaboradores[Matricula],
        "Polo_Base",           "Matriz",
        "Polo Base (Regra)",   "Matriz",
        "IsMatriz",            TRUE(),
        "Grupo",               "Gestor",
        "Fonte",               "DIM (Matriz)"
    )

VAR _leaders_os_raw =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(FactOS, "__NomeNorm", UPPER(TRIM(FactOS[Líder])) ),
            NOT ISBLANK(FactOS[Líder]) &&
            [__NomeNorm] <> "MATRIZ"
        ),
        "Nome",  FactOS[Líder],
        "Cargo", "Líder"
    )

VAR _coords_os_raw =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(FactOS, "__NomeNorm", UPPER(TRIM(FactOS[Coordenador.Coordenador])) ),
            NOT ISBLANK(FactOS[Coordenador.Coordenador]) &&
            [__NomeNorm] <> "MATRIZ"
        ),
        "Nome",  FactOS[Coordenador.Coordenador],
        "Cargo", "Coordenador"
    )

VAR _gestores03_raw = DISTINCT( UNION(_leaders_os_raw, _coords_os_raw) )

VAR _gestores03_nm_join =
    ADDCOLUMNS(
        _gestores03_raw,
        "__NomeNorm",  UPPER(TRIM([Nome])),
        "__CargoNorm", SUBSTITUTE( UPPER(TRIM([Cargo])), "Í","I" )
    )

VAR _dim_gestores_matriz_norm =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(
                DimColaboradores,
                "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" ),
                "__EhMatriz",
                    VAR _raw   = FORMAT(DimColaboradores[IsMatriz], "")
                    VAR _norm  = UPPER(TRIM(COALESCE(_raw, "")))
                    VAR _txt   = _norm IN {"TRUE","VERDADEIRO","SIM","YES","1"}
                    VAR _polo  = UPPER(TRIM(COALESCE(DimColaboradores[Polo_Base], "")))
                    RETURN _txt || _polo = "MATRIZ" || _polo = "PAGRESOLVE MATRIZ SP"
            ),
            [__CargoNorm] IN {"LIDER","COORDENADOR"} && [__EhMatriz]
        ),
        "__NomeNorm",  UPPER(TRIM(DimColaboradores[Nome])),
        "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" )
    )

VAR _gestores03_nm_norm_only =
    SELECTCOLUMNS(_gestores03_nm_join, "__NomeNorm", [__NomeNorm], "__CargoNorm", [__CargoNorm])

VAR _gestores03_nm_norm =
    EXCEPT(_gestores03_nm_norm_only, _dim_gestores_matriz_norm)

VAR _gestores03_nm_sel =
    SELECTCOLUMNS(
        NATURALINNERJOIN(_gestores03_nm_norm, _gestores03_nm_join),
        "Nome",                [Nome],
        "Nome_Normalizado",
            VAR n3 = UPPER(TRIM([Nome]))
            VAR n3a = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n3,
                    "Á","A"),"À","A"),"Â","A"),"Ã","A"),"É","E"),"Ê","E"),"Í","I"),"Ó","O"),"Ô","O"),"Õ","O")
            RETURN SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n3a,"Ú","U"),"Ü","U"),"Ç","C"),
        "Cargo",               [Cargo],
        "Matricula",           BLANK(),
        "Polo_Base",           BLANK(),
        "Polo Base (Regra)",   BLANK(),
        "IsMatriz",            FALSE(),
        "Grupo",               "Gestor",
        "Fonte",               "03 (não Matriz)"
    )

VAR _dim_tecs_mats =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(DimColaboradores,
                "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" )
            ),
            [__CargoNorm] = "TECNICO"
        ),
        "Matricula", DimColaboradores[Matricula]
    )

VAR _dim_assistentes_mats =
    SELECTCOLUMNS(
        FILTER(
            ADDCOLUMNS(
                DimColaboradores,
                "__CargoNorm", SUBSTITUTE( UPPER(TRIM(DimColaboradores[Cargo])), "Í","I" )
            ),
            [__CargoNorm] = "ASSISTENTE"
        ),
        "Matricula", DimColaboradores[Matricula]
    )

VAR _mats_fact_os =
    SELECTCOLUMNS(
        FILTER(FactOS, NOT ISBLANK(FactOS[Matricula])),
        "Matricula", FactOS[Matricula]
    )

VAR _mats_fact_csat =
    SELECTCOLUMNS(
        FILTER(FactCSAT, NOT ISBLANK(FactCSAT[Matricula2])),
        "Matricula", FactCSAT[Matricula2]
    )

VAR _mats_fact_all =
    DISTINCT( UNION(_mats_fact_os, _mats_fact_csat) )

VAR _mats_to_add =
    EXCEPT(
        EXCEPT(_mats_fact_all, _dim_tecs_mats),
        _dim_assistentes_mats
    )

VAR _tecnicos_fatos_add_base =
    ADDCOLUMNS(
        _mats_to_add,
        "__NomeDIM", LOOKUPVALUE(DimColaboradores[Nome],      DimColaboradores[Matricula], [Matricula]),
        "__PoloDIM", LOOKUPVALUE(DimColaboradores[Polo_Base], DimColaboradores[Matricula], [Matricula]),
        "__Polo03",
            VAR m = [Matricula]
            VAR t_top =
                CALCULATETABLE(
                    TOPN(
                        1,
                        ADDCOLUMNS(
                            SUMMARIZE(FactOS, FactOS[des_service_provider]),
                            "__Qtd", COUNTROWS(FactOS)
                        ),
                        [__Qtd], DESC
                    ),
                    FactOS[Matricula] = m
                )
            RETURN MAXX(t_top, FactOS[des_service_provider])
    )

VAR _tecnicos_fatos_add =
    SELECTCOLUMNS(
        _tecnicos_fatos_add_base,
        "Nome",                [__NomeDIM],
        "Nome_Normalizado",
            VAR base = COALESCE([__NomeDIM], "MAT-" & [Matricula])
            VAR n4 = UPPER(TRIM(base))
            VAR n4a = SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n4,
                    "Á","A"),"À","A"),"Â","A"),"Ã","A"),"É","E"),"Ê","E"),"Í","I"),"Ó","O"),"Ô","O"),"Õ","O")
            RETURN SUBSTITUTE(SUBSTITUTE(SUBSTITUTE(n4a,"Ú","U"),"Ü","U"),"Ç","C"),
        "Cargo",               "Técnico",
        "Matricula",           [Matricula],
        "Polo_Base",           [__PoloDIM],
        "Polo Base (Regra)",   IF( UPPER(TRIM(COALESCE([__PoloDIM],[__Polo03],""))) IN {"MATRIZ","PAGRESOLVE MATRIZ SP"}, "Matriz", COALESCE([__PoloDIM], [__Polo03]) ),
        "IsMatriz",            FALSE(),
        "Grupo",               "Operação",
        "Fonte",               "Fato (03/04 sem DIM)"
    )

VAR _union_all =
    UNION(
        _ops_dim,
        _gest_matriz_dim,
        _gestores03_nm_sel,
        _tecnicos_fatos_add
    )
RETURN
DISTINCT(_union_all)


--------------------------------------------------------------------------------
2. COLUNA CALCULADA: Coordenador (na tabela Eixo_Colaboradores)
--------------------------------------------------------------------------------
Para MATRIZ: retorna "Matriz" (evita campo vazio, agrupa sob "Matriz").
Para fora da Matriz: busca Coordenador na FactOS pelo polo.

Coordenador =
VAR PoloAtual = [Polo Base (Regra)]
VAR PoloUpper = UPPER(TRIM(COALESCE(PoloAtual, "")))
VAR EhMatriz = [IsMatriz] || PoloUpper = "MATRIZ" || PoloUpper = "PAGRESOLVE MATRIZ SP"
RETURN
    IF(
        EhMatriz,
        "Matriz",
        IF(
            ISBLANK(PoloAtual) || PoloAtual = "",
            BLANK(),
            CALCULATE(
                FIRSTNONBLANK(FactOS[Coordenador.Coordenador], 1),
                FILTER(ALL(FactOS), FactOS[des_service_provider] = PoloAtual)
            )
        )
    )


--------------------------------------------------------------------------------
3. COLUNA CALCULADA: Líder (na tabela Eixo_Colaboradores)
--------------------------------------------------------------------------------
Para MATRIZ: retorna "Matriz".
Para fora da Matriz: busca Líder na FactOS pelo polo.

Líder =
VAR PoloAtual = [Polo Base (Regra)]
VAR PoloUpper = UPPER(TRIM(COALESCE(PoloAtual, "")))
VAR EhMatriz = [IsMatriz] || PoloUpper = "MATRIZ" || PoloUpper = "PAGRESOLVE MATRIZ SP"
RETURN
    IF(
        EhMatriz,
        "Matriz",
        IF(
            ISBLANK(PoloAtual) || PoloAtual = "" || PoloAtual = "Matriz",
            BLANK(),
            CALCULATE(
                FIRSTNONBLANK(FactOS[Líder], 1),
                FILTER(ALL(FactOS), FactOS[des_service_provider] = PoloAtual)
            )
        )
    )


--------------------------------------------------------------------------------
4. COLUNA CALCULADA: Cargo_Padronizado (na tabela Eixo_Colaboradores)
--------------------------------------------------------------------------------
Mapeia Cargo para Coordenador, Líder, Assistente ou Técnico.
Usada na Matriz para o ramo Matriz → Cargo → Nome.

Cargo_Padronizado =
VAR c = SUBSTITUTE( UPPER(TRIM(COALESCE([Cargo], ""))), "Í","I" )
RETURN
    SWITCH(
        TRUE(),
        c = "COORDENADOR", "Coordenador",
        c = "LIDER",       "Líder",
        c = "ASSISTENTE",  "Assistente",
        c = "TECNICO",     "Técnico",
        [Cargo]
    )


--------------------------------------------------------------------------------
5. CONFIGURAÇÃO DA MATRIZ (Visual no Power BI)
--------------------------------------------------------------------------------

Para o ramo MATRIZ (Matriz → Cargo → Nome):
  Linhas: Coordenador | Líder | Polo Base (Regra) | Cargo_Padronizado | Nome
  - Coordenador = "Matriz"  → agrupa todos
  - Líder = "Matriz"       → segundo nível (todos juntos)
  - Polo Base (Regra) = "Matriz"
  - Cargo_Padronizado = Coordenador / Líder / Assistente (subdivisão por cargo)
  - Nome = nome da pessoa

Para o ramo FORA DA MATRIZ (Coordenador → Líder → Polo → Cargo → Nome):
  - Coordenador = nome do coordenador
  - Líder = nome do líder
  - Polo Base (Regra) = polo
  - Cargo_Padronizado = Assistente / Técnico
  - Nome = nome da pessoa

Valores sugeridos: CSAT Médio (Regra Cargo), SLA_% (Regra Cargo), Ativacao_% (Regra Cargo)


--------------------------------------------------------------------------------
6. ALTERAÇÃO PRINCIPAL NO EIXO_COLABORADORES
--------------------------------------------------------------------------------
Em _ops_dim: IsMatriz passou de FALSE() fixo para [__EhMatrizPolo].
Assim, Assistentes e Técnicos cujo Polo_Base é Matriz/PAGRESOLVE MATRIZ SP
ficam com IsMatriz = TRUE, e as colunas Coordenador e Líder retornam "Matriz"
corretamente para eles também.
